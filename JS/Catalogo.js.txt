document.addEventListener("DOMContentLoaded", () => {
  const containerProdutos = document.getElementById("lista-produtos");
  const inputBusca = document.getElementById("input-busca");
  const botoesCategoria = document.querySelectorAll(".btn-categoria");

  let categoriaAtual = "TODOS";
  let termoBusca = "";

  function renderizarProdutos(lista) {
    if (!containerProdutos) return;
    
    if (lista.length === 0) {
      containerProdutos.innerHTML = "<p class='sem-produtos'>Nenhum produto encontrado.</p>";
      return;
    }

    containerProdutos.innerHTML = lista.map(prod => `
      <div class="card-produto" data-id="${prod.id}">
        <div class="card-header">
          <span class="badge ${prod.badge === 'SOB ENCOMENDA' ? 'encomenda' : 'pronta-entrega'}">
            ${prod.badge}
          </span>
          <img src="${prod.imagem}" alt="${prod.nome}" class="img-produto" />
        </div>
        <div class="card-body">
          <span class="categoria-label">${prod.categoria}</span>
          <h3 class="titulo-produto">${prod.nome}</h3>
          <p class="descricao-produto">${prod.descricao}</p>
          <ul class="detalhes-produto">
            <li><strong>Embalagem:</strong> ${prod.embalagem}</li>
            <li>${prod.caixa}</li>
            <li><strong>Validade:</strong> ${prod.validade}</li>
          </ul>
          <div class="card-acoes">
            <button class="btn-adicionar" onclick="adicionarAoPedido('${prod.nome}')">
              Adicionar ao pedido
            </button>
            <button class="btn-whatsapp" onclick="enviarWhatsApp('${prod.nome}')">
              WhatsApp
            </button>
          </div>
        </div>
      </div>
    `).join("");
  }

  function filtrarProdutos() {
    let filtrados = produtos;

    if (categoriaAtual !== "TODOS") {
      filtrados = filtrados.filter(p => p.categoria.toUpperCase() === categoriaAtual.toUpperCase());
    }

    if (termoBusca.trim() !== "") {
      filtrados = filtrados.filter(p => 
        p.nome.toLowerCase().includes(termoBusca.toLowerCase()) || 
        p.descricao.toLowerCase().includes(termoBusca.toLowerCase())
      );
    }

    renderizarProdutos(filtrados);
  }

  if (inputBusca) {
    inputBusca.addEventListener("input", (e) => {
      termoBusca = e.target.value;
      filtrarProdutos();
    });
  }

  botoesCategoria.forEach(btn => {
    btn.addEventListener("click", () => {
      botoesCategoria.forEach(b => b.classList.remove("active"));
      btn.classList.add("active");
      categoriaAtual = btn.dataset.categoria || "TODOS";
      filtrarProdutos();
    });
  });

  renderizarProdutos(produtos);
});

function adicionarAoPedido(nomeProduto) {
  let carrinho = JSON.parse(localStorage.getItem("carrinhoGR")) || [];
  carrinho.push(nomeProduto);
  localStorage.setItem("carrinhoGR", JSON.stringify(carrinho));
  alert(`"${nomeProduto}" foi adicionado ao seu pedido!`);
  
  const contador = document.getElementById("contador-carrinho");
  if(contador) contador.innerText = carrinho.length;
}

function enviarWhatsApp(nomeProduto) {
  const numeroTelefone = "5500000000000";
  const mensagem = encodeURIComponent(`Olá, gostaria de solicitar um orçamento para o produto: ${nomeProduto}`);
  window.open(`https://wa.me/${numeroTelefone}?text=${mensagem}`, "_blank");
}
